# Treasure of Time MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an installable Android APK of the Treasure of Time MVP — email/password auth, onboarding (pick 2-3 seed tasks per category), and a Home screen where marking tasks complete accumulates and displays recovered time.

**Architecture:** Expo (managed) + TypeScript app. A `RootNavigator` picks between Auth / Onboarding / Home+Profile stacks based on Firebase Auth state and whether the user has any active tasks in Firestore (our "onboarded" signal — the spec's data model has no separate `onboardingCompleted` flag, so we derive it from `tasks` subcollection non-emptiness rather than add a field the spec doesn't define). Firestore repositories are thin wrapper functions; a pure `calculateTotalMinutes` function is the one piece of logic under unit test, per the spec's minimal testing strategy.

**Tech Stack:** Expo (managed workflow) + TypeScript, React Navigation (native-stack), Firebase Auth (email/password) + Firestore (modular v9+ SDK), React Context for auth state, `jest-expo` for the one unit-tested module. npm as package manager.

## Global Constraints

- Onboarding enforces max 3 tasks per category (spec: "Enforces the 'don't overwhelm' MVP principle... max 3 tasks per category"); this plan also enforces a minimum of 2, per the SOW's "2–3 tareas por categoría".
- No Redux — React Context + hooks only ("app state is small enough that a global store would add ceremony without benefit").
- Testing is intentionally minimal: only `calculateTotalMinutes` gets unit tests (1-2 cases). All screens are verified by manual QA (`npx expo start`), not automated tests. No e2e/UI test automation in this phase.
- Data model is exactly:
  - `users/{uid}`: `email`, `displayName`, `createdAt`
  - `users/{uid}/tasks/{taskId}`: `category` (`'health' | 'home' | 'work'`), `title`, `minutesValue`, `isActive`
  - `users/{uid}/completions/{completionId}`: `taskId`, `completedAt`, `minutesEarned`
- Total recovered time is always derived by summing `completions`, never stored as a standalone counter.
- Out of scope: Google Sign-In, Suggestion system, Protected Mode, Time Bank/personal activities, emotional closure/reflection, metrics tracking (Jira epics `TTA-9`–`TTA-12`).

- [ ] **Step 1: Create `src/contexts/AuthContext.tsx`**

```tsx
import React, { createContext, ReactNode, useContext, useEffect, useState } from 'react';
import { onAuthStateChanged, User } from 'firebase/auth';
import { auth } from '../firebase/config';

interface AuthContextValue {
  user: User | null;
  initializing: boolean;
}

const AuthContext = createContext<AuthContextValue>({ user: null, initializing: true });

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [initializing, setInitializing] = useState(true);

  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, (firebaseUser) => {
      setUser(firebaseUser);
      setInitializing(false);
    });
    return unsubscribe;
  }, []);

  return (
    <AuthContext.Provider value={{ user, initializing }}>{children}</AuthContext.Provider>
  );
}

export function useAuth(): AuthContextValue {
  return useContext(AuthContext);
}
```

- [ ] **Step 1: Create `src/data/tasksRepository.ts`**

```typescript
import { collection, doc, getDocs, query, setDoc, where } from 'firebase/firestore';
import { db } from '../firebase/config';
import { UserTask } from '../types/models';

export async function getUserTasks(uid: string): Promise<UserTask[]> {
  const tasksRef = collection(db, 'users', uid, 'tasks');
  const activeQuery = query(tasksRef, where('isActive', '==', true));
  const snapshot = await getDocs(activeQuery);
  return snapshot.docs.map((docSnap) => ({
    id: docSnap.id,
    ...(docSnap.data() as Omit<UserTask, 'id'>),
  }));
}

export async function addUserTask(uid: string, task: Omit<UserTask, 'id'>): Promise<void> {
  const taskRef = doc(collection(db, 'users', uid, 'tasks'));
  await setDoc(taskRef, task);
}
```

- [ ] **Step 2: Create `src/data/completionsRepository.ts`**

```typescript
import { addDoc, collection, getDocs, serverTimestamp, Timestamp } from 'firebase/firestore';
import { db } from '../firebase/config';
import { Completion } from '../types/models';

export async function addCompletion(
  uid: string,
  taskId: string,
  minutesEarned: number
): Promise<void> {
  const completionsRef = collection(db, 'users', uid, 'completions');
  await addDoc(completionsRef, {
    taskId,
    minutesEarned,
    completedAt: serverTimestamp(),
  });
}

export async function getTodayCompletions(uid: string): Promise<Completion[]> {
  const completionsRef = collection(db, 'users', uid, 'completions');
  const snapshot = await getDocs(completionsRef);
  const startOfToday = new Date();
  startOfToday.setHours(0, 0, 0, 0);

  return snapshot.docs
    .map((docSnap) => {
      const data = docSnap.data();
      const completedAt: Date = (data.completedAt as Timestamp | undefined)?.toDate() ?? new Date();
      return {
        id: docSnap.id,
        taskId: data.taskId as string,
        minutesEarned: data.minutesEarned as number,
        completedAt: completedAt.getTime(),
      };
    })
    .filter((completion) => completion.completedAt >= startOfToday.getTime());
}
```

```tsx
import React, { useEffect, useState } from 'react';
import { ActivityIndicator, View } from 'react-native';
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import { useAuth } from '../contexts/AuthContext';
import { getUserTasks } from '../data/tasksRepository';
import { AuthScreen } from '../screens/AuthScreen';
import { OnboardingScreen } from '../screens/OnboardingScreen';
import { HomeScreen } from '../screens/HomeScreen';
import { ProfileScreen } from '../screens/ProfileScreen';
import { RootStackParamList } from './types';

const Stack = createNativeStackNavigator<RootStackParamList>();

export function RootNavigator() {
  const { user, initializing } = useAuth();
  const [checkingOnboarding, setCheckingOnboarding] = useState(true);
  const [onboarded, setOnboarded] = useState(false);

  useEffect(() => {
    if (!user) {
      setCheckingOnboarding(false);
      setOnboarded(false);
      return;
    }
    setCheckingOnboarding(true);
    getUserTasks(user.uid)
      .then((tasks) => setOnboarded(tasks.length > 0))
      .finally(() => setCheckingOnboarding(false));
  }, [user]);

  if (initializing || checkingOnboarding) {
    return (
      <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center' }}>
        <ActivityIndicator size="large" />
      </View>
    );
  }

  return (
    <NavigationContainer>
      <Stack.Navigator screenOptions={{ headerShown: false }}>
        {!user ? (
          <Stack.Screen name="Auth" component={AuthScreen} />
        ) : !onboarded ? (
          <Stack.Screen name="Onboarding">
            {() => <OnboardingScreen onOnboarded={() => setOnboarded(true)} />}
          </Stack.Screen>
        ) : (
          <>
            <Stack.Screen name="Home" component={HomeScreen} />
            <Stack.Screen name="Profile" component={ProfileScreen} />
          </>
        )}
      </Stack.Navigator>
    </NavigationContainer>
  );
}
```

```tsx
import React, { useState } from 'react';
import { Alert, Pressable, ScrollView, StyleSheet, Text, View } from 'react-native';
import { useAuth } from '../contexts/AuthContext';
import { addUserTask } from '../data/tasksRepository';
import { CATEGORY_LABELS, TASK_CATALOG } from '../data/taskCatalog';
import { Category } from '../types/models';

const CATEGORIES: Category[] = ['health', 'home', 'work'];
const MAX_PER_CATEGORY = 3;
const MIN_PER_CATEGORY = 2;

interface Props {
  onOnboarded: () => void;
}

export function OnboardingScreen({ onOnboarded }: Props) {
  const { user } = useAuth();
  const [selected, setSelected] = useState<Set<string>>(new Set());
  const [saving, setSaving] = useState(false);

  function selectedCountFor(category: Category): number {
    return TASK_CATALOG.filter((task) => task.category === category && selected.has(task.id)).length;
  }

  function toggleTask(taskId: string, category: Category) {
    setSelected((prev) => {
      const next = new Set(prev);
      if (next.has(taskId)) {
        next.delete(taskId);
        return next;
      }
      if (selectedCountFor(category) >= MAX_PER_CATEGORY) {
        return prev;
      }
      next.add(taskId);
      return next;
    });
  }

  async function handleConfirm() {
    if (!user) return;
    const missingCategory = CATEGORIES.find((category) => selectedCountFor(category) < MIN_PER_CATEGORY);
    if (missingCategory) {
      Alert.alert('Elegí al menos 2 tareas', `Falta elegir tareas en ${CATEGORY_LABELS[missingCategory]}.`);
      return;
    }
    setSaving(true);
    const chosenTasks = TASK_CATALOG.filter((task) => selected.has(task.id));
    await Promise.all(
      chosenTasks.map((task) =>
        addUserTask(user.uid, {
          category: task.category,
          title: task.title,
          minutesValue: task.minutesValue,
          isActive: true,
        })
      )
    );
    setSaving(false);
    onOnboarded();
  }

  return (
    <ScrollView style={styles.container} contentContainerStyle={{ padding: 24 }}>
      <Text style={styles.title}>Elegí tus primeras tareas</Text>
      <Text style={styles.subtitle}>2 a 3 por categoría. Esto sí lo podés sostener.</Text>
      {CATEGORIES.map((category) => (
        <View key={category} style={styles.categoryBlock}>
          <Text style={styles.categoryTitle}>{CATEGORY_LABELS[category]}</Text>
          {TASK_CATALOG.filter((task) => task.category === category).map((task) => {
            const isSelected = selected.has(task.id);
            return (
              <Pressable
                key={task.id}
                style={[styles.taskRow, isSelected && styles.taskRowSelected]}
                onPress={() => toggleTask(task.id, category)}
              >
                <Text>{task.title}</Text>
                <Text style={styles.minutes}>+{task.minutesValue} min</Text>
              </Pressable>
            );
          })}
        </View>
      ))}
      <Pressable style={styles.confirmButton} onPress={handleConfirm} disabled={saving}>
        <Text style={styles.confirmButtonText}>{saving ? 'Guardando...' : 'Empezar'}</Text>
      </Pressable>
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1 },
  title: { fontSize: 22, fontWeight: '600', marginBottom: 4 },
  subtitle: { color: '#666', marginBottom: 24 },
  categoryBlock: { marginBottom: 24 },
  categoryTitle: { fontSize: 16, fontWeight: '600', marginBottom: 8 },
  taskRow: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    padding: 12,
    borderWidth: 1,
    borderColor: '#ddd',
    borderRadius: 8,
    marginBottom: 8,
  },
  taskRowSelected: { borderColor: '#2d6a4f', backgroundColor: '#eaf4ee' },
  minutes: { color: '#2d6a4f', fontWeight: '600' },
  confirmButton: { backgroundColor: '#2d6a4f', borderRadius: 8, padding: 14, alignItems: 'center', marginTop: 8 },
  confirmButtonText: { color: '#fff', fontWeight: '600' },
});
```

```tsx
import React, { useCallback, useRef, useState } from 'react';
import { Animated, Pressable, ScrollView, StyleSheet, Text, View } from 'react-native';
import { useFocusEffect, useNavigation } from '@react-navigation/native';
import { NativeStackNavigationProp } from '@react-navigation/native-stack';
import { useAuth } from '../contexts/AuthContext';
import { getUserTasks } from '../data/tasksRepository';
import { addCompletion, getTodayCompletions } from '../data/completionsRepository';
import { calculateTotalMinutes } from '../domain/recoveredTime';
import { CATEGORY_LABELS } from '../data/taskCatalog';
import { Category, UserTask } from '../types/models';
import { RootStackParamList } from '../navigation/types';

const CATEGORIES: Category[] = ['health', 'home', 'work'];

export function HomeScreen() {
  const { user } = useAuth();
  const navigation = useNavigation<NativeStackNavigationProp<RootStackParamList>>();
  const [tasks, setTasks] = useState<UserTask[]>([]);
  const [completedTaskIds, setCompletedTaskIds] = useState<Set<string>>(new Set());
  const [totalMinutes, setTotalMinutes] = useState(0);
  const [loading, setLoading] = useState(true);
  const pulseAnim = useRef(new Animated.Value(0)).current;

  const loadData = useCallback(async () => {
    if (!user) return;
    setLoading(true);
    const [userTasks, completions] = await Promise.all([
      getUserTasks(user.uid),
      getTodayCompletions(user.uid),
    ]);
    setTasks(userTasks);
    setCompletedTaskIds(new Set(completions.map((completion) => completion.taskId)));
    setTotalMinutes(calculateTotalMinutes(completions));
    setLoading(false);
  }, [user]);

  useFocusEffect(
    useCallback(() => {
      loadData();
    }, [loadData])
  );

  function playPulse() {
    pulseAnim.setValue(0);
    Animated.sequence([
      Animated.timing(pulseAnim, { toValue: 1, duration: 150, useNativeDriver: true }),
      Animated.timing(pulseAnim, { toValue: 0, duration: 350, useNativeDriver: true }),
    ]).start();
  }

  async function handleComplete(task: UserTask) {
    if (!user || completedTaskIds.has(task.id)) return;
    setCompletedTaskIds((prev) => new Set(prev).add(task.id));
    setTotalMinutes((prev) => prev + task.minutesValue);
    playPulse();
    await addCompletion(user.uid, task.id, task.minutesValue);
  }

  const scale = pulseAnim.interpolate({ inputRange: [0, 1], outputRange: [1, 1.15] });

  if (loading) {
    return (
      <View style={styles.center}>
        <Text>Cargando...</Text>
      </View>
    );
  }

  return (
    <ScrollView style={styles.container} contentContainerStyle={{ padding: 24 }}>
      <View style={styles.header}>
        <Animated.Text style={[styles.total, { transform: [{ scale }] }]}>
          Hoy recuperaste {totalMinutes} minutos para vos.
        </Animated.Text>
        <Pressable onPress={() => navigation.navigate('Profile')}>
          <Text style={styles.profileLink}>Perfil</Text>
        </Pressable>
      </View>
      {CATEGORIES.map((category) => {
        const categoryTasks = tasks.filter((task) => task.category === category);
        if (categoryTasks.length === 0) return null;
        return (
          <View key={category} style={styles.categoryBlock}>
            <Text style={styles.categoryTitle}>{CATEGORY_LABELS[category]}</Text>
            {categoryTasks.map((task) => {
              const done = completedTaskIds.has(task.id);
              return (
                <Pressable
                  key={task.id}
                  style={[styles.taskRow, done && styles.taskRowDone]}
                  onPress={() => handleComplete(task)}
                  disabled={done}
                >
                  <Text style={done ? styles.taskTitleDone : undefined}>{task.title}</Text>
                  <Text style={styles.minutes}>+{task.minutesValue} min</Text>
                </Pressable>
              );
            })}
          </View>
        );
      })}
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1 },
  center: { flex: 1, justifyContent: 'center', alignItems: 'center' },
  header: { marginBottom: 24 },
  total: { fontSize: 20, fontWeight: '600', marginBottom: 8 },
  profileLink: { color: '#2d6a4f' },
  categoryBlock: { marginBottom: 24 },
  categoryTitle: { fontSize: 16, fontWeight: '600', marginBottom: 8 },
  taskRow: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    padding: 12,
    borderWidth: 1,
    borderColor: '#ddd',
    borderRadius: 8,
    marginBottom: 8,
  },
  taskRowDone: { backgroundColor: '#eaf4ee', borderColor: '#2d6a4f' },
  taskTitleDone: { textDecorationLine: 'line-through', color: '#888' },
  minutes: { color: '#2d6a4f', fontWeight: '600' },
});
```
